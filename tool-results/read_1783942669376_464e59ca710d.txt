    41→async function parseTabularFile(file: File): Promise<Record<string, unknown>[]> {
    42→  const name = file.name.toLowerCase();
    43→  if (name.endsWith('.csv')) {
    44→    const text = await file.text();
    45→    const parsed = Papa.parse(text, { header: true, skipEmptyLines: true, dynamicTyping: true });
    46→    return parsed.data as Record<string, unknown>[];
    47→  } else {
    48→    const buf = await file.arrayBuffer();
    49→    const wb = new ExcelJS.Workbook();
    50→    await wb.xlsx.load(buf);
    51→    const ws = wb.worksheets[0];
    52→    const rows: Record<string, unknown>[] = [];
    53→    let headers: string[] = [];
    54→    ws.eachRow((row, rowNumber) => {
    55→      const values = row.values.slice(1);
    56→      if (rowNumber === 1) {
    57→        headers = values.map(v => (v === undefined || v === null) ? '' : String(v).trim());
    58→      } else {
    59→        const obj: Record<string, unknown> = {};
    60→        headers.forEach((h, i) => {
    61→          let v = values[i];
    62→          if (v && typeof v === 'object' && 'text' in (v as object)) v = (v as { text: unknown }).text;
    63→          if (v && typeof v === 'object' && 'result' in (v as object)) v = (v as { result: unknown }).result;
    64→          obj[h] = v === undefined ? null : v;
    65→        });
    66→        rows.push(obj);
    67→      }
    68→    });
    69→    return rows;
    70→  }
    71→}

    73→function esc(v: unknown): string {
    74→  if (v === null || v === undefined) return '';
    75→  return String(v).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]!));
    76→}

   693→  const [cat, setCat] = useState(up(suggestion.categoria || ''));

   702→    <div className="glass-card rounded-xl overflow-hidden">
   703→      <div className="p-4">

   706→          <Checkbox checked={checked} onCheckedChange={(v) => setChecked(!!v)} className="data-[state=checked]:bg-amber-500 data-[state=checked]:border-amber-500" />
   707→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   708→            <Input value={cat} onChange={e => setCat(up(e.target.value))} placeholder="CATEGORÍA" className="font-mono text-xs bg-white/[0.03] border-white/[0.06] h-8 rounded-lg" />
   709→            <Input value={syns} onChange={e => setSyns(e.target.value)} placeholder="Sinónimos separados por coma" className="font-mono text-xs bg-white/[0.03] border-white/[0.06] h-8 rounded-lg" />
   710→          </div>
   711→        </div>
   712→      </div>
   713→    </div>
   714→  );
