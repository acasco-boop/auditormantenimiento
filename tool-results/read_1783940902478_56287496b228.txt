    41→function loadCustomDict(): Record<string, string[]> {
    42→  if (typeof window === 'undefined') return {};
    43→  try {
    44→    const raw = localStorage.getItem(CUSTOM_DICT_KEY);
    45→    return raw ? JSON.parse(raw) : {};
    46→  } catch { return {}; }

    53→async function parseTabularFile(file: File): Promise<Record<string, unknown>[]> {
    54→  const name = file.name.toLowerCase();
    55→  if (name.endsWith('.csv')) {
    56→    const text = await file.text();
    57→    const parsed = Papa.parse(text, { header: true, skipEmptyLines: true, dynamicTyping: true });
    58→    return parsed.data as Record<string, unknown>[];
    59→  } else {
    60→    const buf = await file.arrayBuffer();
    61→    const wb = new ExcelJS.Workbook();
    62→    await wb.xlsx.load(buf);
    63→    const ws = wb.worksheets[0];
    64→    const rows: Record<string, unknown>[] = [];
    65→    let headers: string[] = [];
    66→    ws.eachRow((row, rowNumber) => {
    67→      const values = row.values.slice(1);
    68→      if (rowNumber === 1) {
    69→        headers = values.map(v => (v === undefined || v === null) ? '' : String(v).trim());
    70→      } else {
    71→        const obj: Record<string, unknown> = {};
    72→        headers.forEach((h, i) => {
    73→          let v = values[i];
    74→          if (v && typeof v === 'object' && 'text' in (v as object)) v = (v as { text: unknown }).text;
    75→          if (v && typeof v === 'object' && 'result' in (v as object)) v = (v as { result: unknown }).result;
    76→          obj[h] = v === undefined ? null : v;
    77→        });
    78→        rows.push(obj);
    79→      }
    80→    });
    81→    return rows;
    82→  }
    83→}

    85→function esc(v: unknown): string {
    86→  if (v === null || v === undefined) return '';
    87→  return String(v).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]!));
    88→}

    90→function thinBorder() {
    91→  const side = { style: 'thin' as const, color: { argb: 'FFD9D9D9' } };
    92→  return { top: side, bottom: side, left: side, right: side };
    93→}

   898→        <div className="grid grid-cols-[auto_1fr] gap-3 items-center">
   899→          <Checkbox checked={checked} onCheckedChange={(v) => setChecked(!!v)} className="data-[state=checked]:bg-amber-500 data-[state=checked]:border-amber-500" />
   900→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   901→            <Input
   902→              value={cat}
   903→              onChange={e => setCat(up(e.target.value))}
   904→              placeholder="CATEGORÍA"

   907→            <Input
   908→              value={syns}
   909→              onChange={e => setSyns(e.target.value)}
   910→              placeholder="Sinónimos separados por coma"
   911→              className="font-mono text-xs bg-slate-900/50 border-slate-700/50 h-8"
   912→            />
   913→          </div>
