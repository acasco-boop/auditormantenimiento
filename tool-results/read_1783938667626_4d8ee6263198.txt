   503→   503→function runAudit(){

   505→   505→  errBox.innerHTML = '';
   506→   506→
   507→   507→  if(!dfTar || !dfMat){ return; }

   515→   515→  if(!okMat.ok){

   519→   519→
   520→   520→  // Materiales indexados por Nro. Orden (global para desglose en métricas)
   521→   521→  matByOrder = {};
   522→   522→  dfMat.forEach(r=>{
   523→   523→    const key = String(r['Nro. Orden']);
   524→   524→    if(!matByOrder[key]) matByOrder[key] = [];
   525→   525→    matByOrder[key].push(r);
   526→   526→  });
   527→   527→
   528→   528→  const activeParts = getActiveParts();
   529→   529→  const unrecognizedMap = {};
   530→   530→
   531→   531→  results = [];
   532→   532→  dfTar.forEach(row=>{
   533→   533→    const tarea = row['Tarea'] === null || row['Tarea'] === undefined ? '' : String(row['Tarea']);
   534→   534→    const tareaUp = up(tarea);
   535→   535→    if(!explicitReplacementNeeded(tareaUp)) return;
   536→   536→
   537→   537→    const order = row['Nro. Orden'];
   538→   538→
   539→   539→    // Detección de repuestos aún no reconocidos por el diccionario (de fábrica
   540→   540→    // + aprendido). Se registra independientemente del cruce con materiales,
   541→   541→    // porque el objetivo es identificar vocabulario nuevo para poder sumarlo.
   542→   542→    const matchesKnownCategory = Object.keys(activeParts).some(kw => tareaUp.includes(kw));
   543→   543→    if(!matchesKnownCategory){
   544→   544→      const key = tareaUp;
   545→   545→      if(!unrecognizedMap[key]) unrecognizedMap[key] = {tarea, count:0, order};
   546→   546→      unrecognizedMap[key].count++;
   547→   547→    }
   548→   548→
   549→   549→    const mats = matByOrder[String(order)] || [];
   550→   550→    const matList = mats.map(m => up(m['Desc. Artículo']));
   551→   551→    const totalSalidas = mats.reduce((acc,m)=>{
   552→   552→      const v = parseFloat(m['Salidas']);
   553→   553→      return acc + (isNaN(v) ? 0 : v);
   554→   554→    }, 0);
   555→   555→
   556→   556→    const base = {
   557→   557→      'Nro. Orden': order,
   558→   558→      'Equipo': row['Codigo equipo'],
   559→   559→      'Nombre Equipo': row['Nombre Equipo'],
   560→   560→      'Tarea': tarea,
   561→   561→      'Estado Tarea': row['Estado']
   562→   562→    };
   563→   563→
   564→   564→    if(mats.length === 0){
   565→   565→      results.push({...base,
   566→   566→        'Tipo de Hallazgo': '1) Orden sin repuestos asignados',
   567→   567→        'Detalle': 'Ningún material cargado en la orden'
   568→   568→      });
   569→   569→    } else if(totalSalidas === 0){
   570→   570→      results.push({...base,
   571→   571→        'Tipo de Hallazgo': '3) Planificado sin Salida física (Salidas = 0)',
   572→   572→        'Detalle': 'Materiales planificados con Salida Cero: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   573→   573→      });
   574→   574→    } else {
   575→   575→      // Se revisan TODAS las categorías mencionadas en la tarea (no se corta
   576→   576→      // en la primera que matchea), así una tarea que pide varios repuestos
   577→   577→      // (ej. "Cambiar filtro y correa") no queda "aprobada" con que solo uno
   578→   578→      // de los dos haya sido cargado en el pañol.
   579→   579→      const missingCategories = [];
   580→   580→      for(const kw in activeParts){
   581→   581→        if(tareaUp.includes(kw)){
   582→   582→          const matsKws = activeParts[kw];
   583→   583→          const found = matList.some(m => matsKws.some(mk => m.includes(mk)));
   584→   584→          if(!found) missingCategories.push(kw);
   585→   585→        }
   586→   586→      }
   587→   587→      if(missingCategories.length > 0){
   588→   588→        results.push({...base,
   589→   589→          'Tipo de Hallazgo': `2) Desconexión de material (Falta '${missingCategories.join("', '")}')`,
   590→   590→          'Detalle': 'Materiales cargados no coinciden: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   591→   591→        });
   592→   592→      }
   593→   593→    }

   599→   599→  renderUnrecognized();
   600→   600→  renderCustomDictList();
   601→   601→}
