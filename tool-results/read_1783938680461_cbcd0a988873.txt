   134→   503→function runAudit(){
   135→   504→  const errBox = document.getElementById('load-status');
   136→   505→  errBox.innerHTML = '';
   137→   506→
   138→   507→  if(!dfTar || !dfMat){ return; }
   139→   508→
   140→   509→  const okTar = normalizeOrderCol(dfTar, 'DocNum');
   141→   510→  if(!okTar.ok){
   142→   511→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (DocNum o Nro. Orden) en el archivo de Tareas.</div>';
   143→   512→    return;
   144→   513→  }
   145→   514→  const okMat = normalizeOrderCol(dfMat, 'Nro. OM');
   146→   515→  if(!okMat.ok){
   147→   516→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (Nro. OM o Nro. Orden) en el archivo de Materiales.</div>';
   148→   517→    return;
   149→   518→  }
   150→   519→
   151→   520→  // Materiales indexados por Nro. Orden (global para desglose en métricas)
   152→   521→  matByOrder = {};
   153→   522→  dfMat.forEach(r=>{
   154→   523→    const key = String(r['Nro. Orden']);
   155→   524→    if(!matByOrder[key]) matByOrder[key] = [];
   156→   525→    matByOrder[key].push(r);
   157→   526→  });
   158→   527→
   159→   528→  const activeParts = getActiveParts();
   160→   529→  const unrecognizedMap = {};
   161→   530→
   162→   531→  results = [];
   163→   532→  dfTar.forEach(row=>{
   164→   533→    const tarea = row['Tarea'] === null || row['Tarea'] === undefined ? '' : String(row['Tarea']);
   165→   534→    const tareaUp = up(tarea);
   166→   535→    if(!explicitReplacementNeeded(tareaUp)) return;
   167→   536→
   168→   537→    const order = row['Nro. Orden'];
   169→   538→
   170→   539→    // Detección de repuestos aún no reconocidos por el diccionario (de fábrica
   171→   540→    // + aprendido). Se registra independientemente del cruce con materiales,
   172→   541→    // porque el objetivo es identificar vocabulario nuevo para poder sumarlo.
   173→   542→    const matchesKnownCategory = Object.keys(activeParts).some(kw => tareaUp.includes(kw));
   174→   543→    if(!matchesKnownCategory){
   175→   544→      const key = tareaUp;
   176→   545→      if(!unrecognizedMap[key]) unrecognizedMap[key] = {tarea, count:0, order};
   177→   546→      unrecognizedMap[key].count++;
   178→   547→    }
   179→   548→
   180→   549→    const mats = matByOrder[String(order)] || [];
   181→   550→    const matList = mats.map(m => up(m['Desc. Artículo']));
   182→   551→    const totalSalidas = mats.reduce((acc,m)=>{
   183→   552→      const v = parseFloat(m['Salidas']);
   184→   553→      return acc + (isNaN(v) ? 0 : v);
   185→   554→    }, 0);
   186→   555→
   187→   556→    const base = {
   188→   557→      'Nro. Orden': order,
   189→   558→      'Equipo': row['Codigo equipo'],
   190→   559→      'Nombre Equipo': row['Nombre Equipo'],
   191→   560→      'Tarea': tarea,
   192→   561→      'Estado Tarea': row['Estado']
   193→   562→    };
   194→   563→
   195→   564→    if(mats.length === 0){
   196→   565→      results.push({...base,
   197→   566→        'Tipo de Hallazgo': '1) Orden sin repuestos asignados',
   198→   567→        'Detalle': 'Ningún material cargado en la orden'
   199→   568→      });
   200→   569→    } else if(totalSalidas === 0){
   201→   570→      results.push({...base,
   202→   571→        'Tipo de Hallazgo': '3) Planificado sin Salida física (Salidas = 0)',
   203→   572→        'Detalle': 'Materiales planificados con Salida Cero: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   204→   573→      });
   205→   574→    } else {
   206→   575→      // Se revisan TODAS las categorías mencionadas en la tarea (no se corta
   207→   576→      // en la primera que matchea), así una tarea que pide varios repuestos
   208→   577→      // (ej. "Cambiar filtro y correa") no queda "aprobada" con que solo uno
   209→   578→      // de los dos haya sido cargado en el pañol.
   210→   579→      const missingCategories = [];
   211→   580→      for(const kw in activeParts){
   212→   581→        if(tareaUp.includes(kw)){
   213→   582→          const matsKws = activeParts[kw];
   214→   583→          const found = matList.some(m => matsKws.some(mk => m.includes(mk)));
   215→   584→          if(!found) missingCategories.push(kw);
   216→   585→        }
   217→   586→      }
   218→   587→      if(missingCategories.length > 0){
   219→   588→        results.push({...base,
   220→   589→          'Tipo de Hallazgo': `2) Desconexión de material (Falta '${missingCategories.join("', '")}')`,
   221→   590→          'Detalle': 'Materiales cargados no coinciden: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   222→   591→        });
   223→   592→      }
   224→   593→    }
   225→   594→  });
   226→   595→
   227→   596→  unrecognizedTasks = Object.values(unrecognizedMap).sort((a,b)=>b.count-a.count);
   228→   597→
   229→   598→  renderResults();
   230→   599→  renderUnrecognized();
   231→   600→  renderCustomDictList();
   232→   601→}
