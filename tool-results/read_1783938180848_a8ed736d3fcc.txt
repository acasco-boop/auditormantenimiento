   299→   299→function readFileAsArrayBuffer(file){
   300→   300→  return new Promise((res, rej)=>{
   301→   301→    const r = new FileReader();
   302→   302→    r.onload = ()=>res(r.result);
   303→   303→    r.onerror = rej;
   304→   304→    r.readAsArrayBuffer(file);
   305→   305→  });
   306→   306→}
   307→   307→function readFileAsText(file){
   308→   308→  return new Promise((res, rej)=>{
   309→   309→    const r = new FileReader();
   310→   310→    r.onload = ()=>res(r.result);
   311→   311→    r.onerror = rej;
   312→   312→    r.readAsText(file, 'utf-8');
   313→   313→  });
   314→   314→}

   439→   439→}

   447→   447→  if(!tareaUp) return false;

   468→   468→function runAudit(){
   469→   469→  const errBox = document.getElementById('load-status');
   470→   470→  errBox.innerHTML = '';
   471→   471→
   472→   472→  if(!dfTar || !dfMat){ return; }
   473→   473→
   474→   474→  const okTar = normalizeOrderCol(dfTar, 'DocNum');
   475→   475→  if(!okTar.ok){
   476→   476→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (DocNum o Nro. Orden) en el archivo de Tareas.</div>';
   477→   477→    return;
   478→   478→  }
   479→   479→  const okMat = normalizeOrderCol(dfMat, 'Nro. OM');
   480→   480→  if(!okMat.ok){
   481→   481→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (Nro. OM o Nro. Orden) en el archivo de Materiales.</div>';
   482→   482→    return;
   483→   483→  }
   484→   484→
   485→   485→  // Materiales indexados por Nro. Orden
   486→   486→  const matByOrder = {};
   487→   487→  dfMat.forEach(r=>{
   488→   488→    const key = String(r['Nro. Orden']);
   489→   489→    if(!matByOrder[key]) matByOrder[key] = [];
   490→   490→    matByOrder[key].push(r);
   491→   491→  });
   492→   492→
   493→   493→  const activeParts = getActiveParts();
   494→   494→  const unrecognizedMap = {};
   495→   495→
   496→   496→  results = [];
   497→   497→  dfTar.forEach(row=>{
   498→   498→    const tarea = row['Tarea'] === null || row['Tarea'] === undefined ? '' : String(row['Tarea']);
   499→   499→    const tareaUp = up(tarea);
   500→   500→    if(!explicitReplacementNeeded(tareaUp)) return;
   501→   501→
   502→   502→    const order = row['Nro. Orden'];
   503→   503→
   504→   504→    // Detección de repuestos aún no reconocidos por el diccionario (de fábrica
   505→   505→    // + aprendido). Se registra independientemente del cruce con materiales,
   506→   506→    // porque el objetivo es identificar vocabulario nuevo para poder sumarlo.
   507→   507→    const matchesKnownCategory = Object.keys(activeParts).some(kw => tareaUp.includes(kw));
   508→   508→    if(!matchesKnownCategory){
   509→   509→      const key = tareaUp;
   510→   510→      if(!unrecognizedMap[key]) unrecognizedMap[key] = {tarea, count:0, order};
   511→   511→      unrecognizedMap[key].count++;
   512→   512→    }
   513→   513→
   514→   514→    const mats = matByOrder[String(order)] || [];
   515→   515→    const matList = mats.map(m => up(m['Desc. Artículo']));
   516→   516→    const totalSalidas = mats.reduce((acc,m)=>{
   517→   517→      const v = parseFloat(m['Salidas']);
   518→   518→      return acc + (isNaN(v) ? 0 : v);
   519→   519→    }, 0);
   520→   520→
   521→   521→    const base = {
   522→   522→      'Nro. Orden': order,
   523→   523→      'Equipo': row['Codigo equipo'],
   524→   524→      'Nombre Equipo': row['Nombre Equipo'],
   525→   525→      'Tarea': tarea,
   526→   526→      'Estado Tarea': row['Estado']
   527→   527→    };
   528→   528→
   529→   529→    if(mats.length === 0){
   530→   530→      results.push({...base,
   531→   531→        'Tipo de Hallazgo': '1) Orden sin repuestos asignados',
   532→   532→        'Detalle': 'Ningún material cargado en la orden'
   533→   533→      });
   534→   534→    } else if(totalSalidas === 0){
   535→   535→      results.push({...base,
   536→   536→        'Tipo de Hallazgo': '3) Planificado sin Salida física (Salidas = 0)',
   537→   537→        'Detalle': 'Materiales planificados con Salida Cero: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   538→   538→      });
   539→   539→    } else {
   540→   540→      // Se revisan TODAS las categorías mencionadas en la tarea (no se corta
   541→   541→      // en la primera que matchea), así una tarea que pide varios repuestos
   542→   542→      // (ej. "Cambiar filtro y correa") no queda "aprobada" con que solo uno
   543→   543→      // de los dos haya sido cargado en el pañol.
   544→   544→      const missingCategories = [];
   545→   545→      for(const kw in activeParts){
   546→   546→        if(tareaUp.includes(kw)){
   547→   547→          const matsKws = activeParts[kw];
   548→   548→          const found = matList.some(m => matsKws.some(mk => m.includes(mk)));
   549→   549→          if(!found) missingCategories.push(kw);
   550→   550→        }
   551→   551→      }
   552→   552→      if(missingCategories.length > 0){
   553→   553→        results.push({...base,
   554→   554→          'Tipo de Hallazgo': `2) Desconexión de material (Falta '${missingCategories.join("', '")}')`,
   555→   555→          'Detalle': 'Materiales cargados no coinciden: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   556→   556→        });
   557→   557→      }
   558→   558→    }
   559→   559→  });
   560→   560→
   561→   561→  unrecognizedTasks = Object.values(unrecognizedMap).sort((a,b)=>b.count-a.count);
   562→   562→
   563→   563→  renderResults();
   564→   564→  renderUnrecognized();
   565→   565→  renderCustomDictList();
   566→   566→}
