   299→function readFileAsArrayBuffer(file){
   300→  return new Promise((res, rej)=>{
   301→    const r = new FileReader();
   302→    r.onload = ()=>res(r.result);
   303→    r.onerror = rej;
   304→    r.readAsArrayBuffer(file);
   305→  });
   306→}
   307→function readFileAsText(file){
   308→  return new Promise((res, rej)=>{
   309→    const r = new FileReader();
   310→    r.onload = ()=>res(r.result);
   311→    r.onerror = rej;
   312→    r.readAsText(file, 'utf-8');
   313→  });
   314→}

   316→// Devuelve array de objetos (filas) a partir de un File .xlsx o .csv
   317→async function parseTabularFile(file){
   318→  const name = file.name.toLowerCase();
   319→  if(name.endsWith('.csv')){
   320→    const text = await readFileAsText(file);
   321→    const parsed = Papa.parse(text, {header:true, skipEmptyLines:true, dynamicTyping:true});
   322→    return parsed.data;
   323→  } else {
   324→    const buf = await readFileAsArrayBuffer(file);
   325→    const wb = new ExcelJS.Workbook();
   326→    await wb.xlsx.load(buf);
   327→    const ws = wb.worksheets[0];
   328→    const rows = [];
   329→    let headers = [];
   330→    ws.eachRow((row, rowNumber)=>{
   331→      const values = row.values.slice(1); // exceljs es 1-indexed, values[0] es undefined
   332→      if(rowNumber === 1){
   333→        headers = values.map(v => (v===undefined||v===null) ? '' : String(v).trim());
   334→      } else {
   335→        const obj = {};
   336→        headers.forEach((h, i)=>{
   337→          let v = values[i];
   338→          if(v && typeof v === 'object' && v.text !== undefined) v = v.text; // rich text
   339→          if(v && typeof v === 'object' && v.result !== undefined) v = v.result; // formula
   340→          obj[h] = (v===undefined) ? null : v;
   341→        });
   342→        rows.push(obj);
   343→      }
   344→    });
   345→    return rows;
   346→  }
   347→}

   435→function stripAccents(str){
   436→  return str
   437→    .replace(/[ÁÀÄÂ]/g,'A').replace(/[ÉÈËÊ]/g,'E').replace(/[ÍÌÏÎ]/g,'I')
   438→    .replace(/[ÓÒÖÔ]/g,'O').replace(/[ÚÙÜÛ]/g,'U').replace(/Ñ/g,'N');
   439→}

   446→function explicitReplacementNeeded(tareaUp){
   447→  if(!tareaUp) return false;
   448→  const controlKw = ['CONTROLAR','REVISION','REVISAR','CHEQUEAR','CONTROL'];
   449→  if(controlKw.some(k=>tareaUp.includes(k)) && tareaUp.includes('SELECTORA')) return false;
   450→  const revKw = ['REVISION','REVISAR','CHEQUEAR'];
   451→  if(revKw.some(k=>tareaUp.includes(k))) return false;
   452→  const actionKw = ['CAMBIAR','REEMPLAZAR','COLOCAR','COLOCACION','INSTALAR','CAMBIO'];
   453→  return actionKw.some(k=>tareaUp.includes(k));
   454→}

   456→function normalizeOrderCol(rows, primaryCol, fallbackCol){
   457→  if(rows.length === 0) return {ok:false};
   458→  const cols = Object.keys(rows[0]);
   459→  if(cols.includes(primaryCol)){
   460→    rows.forEach(r=> r['Nro. Orden'] = r[primaryCol]);

   463→    return {ok:true};
   464→  }
   465→  return {ok:false};

   468→function runAudit(){
   469→  const errBox = document.getElementById('load-status');
   470→  errBox.innerHTML = '';
   471→
   472→  if(!dfTar || !dfMat){ return; }
   473→
   474→  const okTar = normalizeOrderCol(dfTar, 'DocNum');
   475→  if(!okTar.ok){
   476→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (DocNum o Nro. Orden) en el archivo de Tareas.</div>';
   477→    return;
   478→  }
   479→  const okMat = normalizeOrderCol(dfMat, 'Nro. OM');
   480→  if(!okMat.ok){
   481→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (Nro. OM o Nro. Orden) en el archivo de Materiales.</div>';
   482→    return;
   483→  }
   484→
   485→  // Materiales indexados por Nro. Orden
   486→  const matByOrder = {};
   487→  dfMat.forEach(r=>{
   488→    const key = String(r['Nro. Orden']);
   489→    if(!matByOrder[key]) matByOrder[key] = [];
   490→    matByOrder[key].push(r);
   491→  });
   492→
   493→  const activeParts = getActiveParts();
   494→  const unrecognizedMap = {};
   495→
   496→  results = [];
   497→  dfTar.forEach(row=>{
   498→    const tarea = row['Tarea'] === null || row['Tarea'] === undefined ? '' : String(row['Tarea']);
   499→    const tareaUp = up(tarea);
   500→    if(!explicitReplacementNeeded(tareaUp)) return;
   501→
   502→    const order = row['Nro. Orden'];
   503→
   504→    // Detección de repuestos aún no reconocidos por el diccionario (de fábrica
   505→    // + aprendido). Se registra independientemente del cruce con materiales,
   506→    // porque el objetivo es identificar vocabulario nuevo para poder sumarlo.
   507→    const matchesKnownCategory = Object.keys(activeParts).some(kw => tareaUp.includes(kw));
   508→    if(!matchesKnownCategory){
   509→      const key = tareaUp;
   510→      if(!unrecognizedMap[key]) unrecognizedMap[key] = {tarea, count:0, order};
   511→      unrecognizedMap[key].count++;
   512→    }
   513→
   514→    const mats = matByOrder[String(order)] || [];
   515→    const matList = mats.map(m => up(m['Desc. Artículo']));
   516→    const totalSalidas = mats.reduce((acc,m)=>{
   517→      const v = parseFloat(m['Salidas']);
   518→      return acc + (isNaN(v) ? 0 : v);
   519→    }, 0);
   520→
   521→    const base = {
   522→      'Nro. Orden': order,
   523→      'Equipo': row['Codigo equipo'],
   524→      'Nombre Equipo': row['Nombre Equipo'],
   525→      'Tarea': tarea,
   526→      'Estado Tarea': row['Estado']
   527→    };
   528→
   529→    if(mats.length === 0){
   530→      results.push({...base,
   531→        'Tipo de Hallazgo': '1) Orden sin repuestos asignados',
   532→        'Detalle': 'Ningún material cargado en la orden'
   533→      });
   534→    } else if(totalSalidas === 0){
   535→      results.push({...base,
   536→        'Tipo de Hallazgo': '3) Planificado sin Salida física (Salidas = 0)',
   537→        'Detalle': 'Materiales planificados con Salida Cero: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   538→      });
   539→    } else {
   540→      // Se revisan TODAS las categorías mencionadas en la tarea (no se corta
   541→      // en la primera que matchea), así una tarea que pide varios repuestos
   542→      // (ej. "Cambiar filtro y correa") no queda "aprobada" con que solo uno
   543→      // de los dos haya sido cargado en el pañol.
   544→      const missingCategories = [];
   545→      for(const kw in activeParts){
   546→        if(tareaUp.includes(kw)){
   547→          const matsKws = activeParts[kw];
   548→          const found = matList.some(m => matsKws.some(mk => m.includes(mk)));
   549→          if(!found) missingCategories.push(kw);
   550→        }
   551→      }
   552→      if(missingCategories.length > 0){
   553→        results.push({...base,
   554→          'Tipo de Hallazgo': `2) Desconexión de material (Falta '${missingCategories.join("', '")}')`,
   555→          'Detalle': 'Materiales cargados no coinciden: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   556→        });
   557→      }
   558→    }
   559→  });
   560→
   561→  unrecognizedTasks = Object.values(unrecognizedMap).sort((a,b)=>b.count-a.count);
   562→
   563→  renderResults();
   564→  renderUnrecognized();
   565→  renderCustomDictList();
   566→}

   829→  const restantes = unrecognizedTasks.length - batch.length;
   830→
   831→  box.innerHTML = '<div class="ai-box loading">🧠 Groq IA analizando repuestos no reconocidos...</div>';
   832→
   833→  const listado = batch.map((u,i)=> `${i+1}. ${u.tarea}`).join('\n');
   834→  const prompt = `Sos un experto en repuestos de mantenimiento de flotas de transporte pesado (camiones, acoplados).
   835→Te paso una lista de tareas de taller que indican un cambio de repuesto, pero cuyo repuesto todavía no está
   836→identificado en un diccionario de categorías.
   837→
   838→Para CADA tarea de la lista, identificá:
   839→- "categoria": el nombre genérico del repuesto que se está cambiando, en UNA sola palabra o frase corta,
   840→  en MAYÚSCULAS, sin tildes (ej: NEUMATICO, RODAMIENTO, BOMBA DE AGUA).
   841→- "sinonimos": una lista de 2 a 5 palabras o frases en MAYÚSCULAS sin tildes que podrían aparecer en la
   842→  descripción de ese mismo repuesto en un sistema de stock de pañol (sinónimos, variantes, nombres comerciales).
   843→
   844→Si dos tareas de la lista se refieren al mismo repuesto, usá la MISMA categoría para ambas.
   845→Si una tarea no menciona ningún repuesto físico reconocible (por ejemplo, es una tarea administrativa o
   846→ambigua), devolvé "categoria": null para esa tarea.
   847→
   848→Respondé ÚNICAMENTE con un array JSON válido, sin texto adicional, sin explicaciones, sin bloques de
   849→markdown, con esta forma exacta:
   850→[{"tarea": "...", "categoria": "...", "sinonimos": ["...", "..."]}, ...]
   851→
   852→Lista de tareas:
   853→${listado}`;
   854→
   855→  try{
   856→    const resp = await fetch('https://api.groq.com/openai/v1/chat/completions', {
   857→      method: 'POST',
   858→      headers: {'Content-Type':'application/json', 'Authorization':'Bearer ' + apiKey},
   859→      body: JSON.stringify({
   860→        model: model,
   861→        messages: [{role:'user', content: prompt}],
   862→        temperature: 0.1,
   863→        max_tokens: Math.min(8192, Math.max(2048, batchSize * 400))
   864→      })
   865→    });
   866→    const data = await resp.json();
   867→    if(!resp.ok){
   868→      const errMsg = (data && data.error && data.error.message) ? data.error.message : JSON.stringify(data);
   869→      if(errMsg.includes('TPM') || errMsg.includes('too large') || errMsg.includes('token')){
   870→        box.innerHTML = `<div class="warn">⚠️ El pedido superó el límite de tokens de tu plan Groq. Se enviaron ${batch.length} tareas — <button class="btn-ghost" style="margin-top:6px;padding:6px 12px;font-size:12px;" onclick="document.getElementById('MAX_TASKS_FALLBACK').value='true';document.getElementById('btn-analizar-dict').click();">Reintentar con lote más chico (${Math.max(3,Math.floor(batch.length/2))} tareas)</button><input type="hidden" id="MAX_TASKS_FALLBACK" value="false">`;
   871→      } else {
   872→        box.innerHTML = `<div class="err">Error al conectar con Groq: ${esc(errMsg)}</div>`;
   873→      }
   874→      return;
   875→    }
   876→    let rawText = data.choices[0].message.content.trim();
   877→    let text = rawText;
   878→
   879→    // === Limpieza robusta multi-capa para respuestas de LLM ===
   880→
   881→    // 1) Quitar bloques de razonamiento  ...  (Qwen3, DeepSeek, etc.)
   882→    text = text.replace(/<think>[\s\S]*?<\/think>/gi, '').trim();
   883→
   884→    // 2) Quitar TODAS las líneas de fence de markdown (```json, ```, etc.)
   885→    text = text.replace(/```[a-zA-Z]*\s*/g, '').replace(/```/g, '').trim();
   886→
   887→    // 3) Extraer solo el array JSON: buscar el primer '[' y el último ']'
   888→    const firstBracket = text.indexOf('[');
   889→    const lastBracket = text.lastIndexOf(']');
   890→    if (firstBracket !== -1 && lastBracket > firstBracket) {
   891→      text = text.substring(firstBracket, lastBracket + 1);
   892→    }
   893→
   894→    // 4) Reemplazar comillas tipográficas por comillas rectas (los LLM a veces las usan)
   895→    text = text.replace(/[\u2018\u2019]/g, "'").replace(/[\u201C\u201D]/g, '"');
   896→
   897→    // 5) Reparar errores comunes del LLM: comas sobrantes antes de } o ]
   898→    text = text.replace(/,\s*([}\]])/g, '$1');
   899→
   900→    // 6) Si el JSON quedó truncado (max_tokens), intentar cerrarlo
   901→    //    Detectar si falta cerrar objetos o el array
   902→    const openBraces = (text.match(/\{/g) || []).length;
   903→    const closeBraces = (text.match(/\}/g) || []).length;
   904→    const openBrackets = (text.match(/\[/g) || []).length;
   905→    const closeBrackets = (text.match(/\]/g) || []).length;
   906→    if (closeBraces < openBraces || closeBrackets < openBrackets) {
   907→      // Quitar la última coma si la hay y cerrar lo que falte
   908→      text = text.replace(/,\s*$/, '');
   909→      const missingBraces = openBraces - closeBraces;
   910→      const missingBrackets = openBrackets - closeBrackets;
   911→      text += '}'.repeat(missingBraces) + ']'.repeat(missingBrackets);
   912→    }
   913→
   914→    let parsed;
   915→    try{ parsed = JSON.parse(text); }
   916→    catch(e){
   917→      box.innerHTML = `<div class="err">La IA no devolvió un JSON válido. Probá analizar de nuevo.
   918→        <br><br><strong style="font-size:12px;">Consejo:</strong> <span style="font-size:12px;">Si usás <em>qwen/qwen3.6-27b</em>, cambiá a <em>llama-3.3-70b-versatile</em> — los modelos de razonamiento a veces incluyen su "pensamiento" en la respuesta y eso rompe el JSON.</span>
   919→        <br><br><span style="font-family:var(--mono);font-size:11px;opacity:.6;">Respuesta cruda (primeros 600 chars):<br>${esc(rawText.slice(0,600))}</span>
   920→        <br><br><span style="font-family:var(--mono);font-size:11px;opacity:.4;">Después de la limpieza:<br>${esc(text.slice(0,600))}</span></div>`;
   921→      return;
   922→    }
   923→    if (!Array.isArray(parsed)) {
   924→      box.innerHTML = `<div class="err">La IA devolvió un JSON válido pero no es un array. Probá analizar de nuevo.<br><span style="font-family:var(--mono);font-size:11px;">${esc(text.slice(0,300))}</span></div>`;
   925→      return;
   926→    }
   927→    if (parsed.length === 0) {
   928→      box.innerHTML = `<div class="warn">La IA devolvió un array vacío. Puede que no haya reconocido repuestos válidos. Probá con otro modelo o revisá las tareas sin categoría.</div>`;
   929→      return;
   930→    }
   931→    const wasTruncated = (data.choices[0].finish_reason === 'length');
   932→    lastSuggestions = parsed.filter(s => s && s.categoria);
   933→    renderSuggestions();
   934→    if(wasTruncated){
   935→      document.getElementById('suggestions-box').insertAdjacentHTML('afterbegin',
   936→        `<div class="warn" style="margin-bottom:10px;">⚠️ La respuesta de la IA se cortó por límite de tokens. Se procesaron ${parsed.length} de ${batch.length} tareas. Guardá estas y volvé a analizar para obtener el resto.</div>`);
   937→    } else if(restantes > 0){
   938→      document.getElementById('suggestions-box').insertAdjacentHTML('beforeend',
   939→        `<div class="warn" style="margin-top:10px;">Se analizaron las primeras ${batch.length} tareas. Quedan ${restantes} más — guardá estas y volvé a analizar para revisar el resto.</div>`);
   940→    }
   941→  }catch(e){
   942→    box.innerHTML = `<div class="err">Error al conectar con Groq: ${esc(e.message)}</div>`;
   943→  }
   944→});

   946→function renderSuggestions(){
   947→  const box = document.getElementById('suggestions-box');
   948→  if(lastSuggestions.length === 0){
   949→    box.innerHTML = '<div class="empty">La IA no identificó repuestos reconocibles en las tareas analizadas.</div>';
   950→    return;
   951→  }
   952→  box.innerHTML = `
   953→    <div class="tag" style="margin-bottom:10px;">Sugerencias de la IA — revisá y ajustá antes de guardar</div>
   954→    ${lastSuggestions.map((s,i)=>`
   955→      <div class="sugg-card">
   956→        <div class="sugg-tarea">"${esc(s.tarea)}"</div>
   957→        <div class="sugg-row">
   958→          <input type="checkbox" id="sugg-check-${i}" checked>
   959→          <input type="text" id="sugg-cat-${i}" value="${esc(up(s.categoria))}" placeholder="CATEGORÍA">
   960→          <input type="text" id="sugg-syn-${i}" value="${esc((s.sinonimos||[]).map(x=>up(x)).join(', '))}" placeholder="Sinónimos separados por coma">
   961→        </div>
   962→      </div>
   963→    `).join('')}
   964→    <button class="btn-primary" id="btn-guardar-sugerencias" style="margin-top:6px;">💾 Guardar seleccionadas en el diccionario</button>
   965→  `;
   966→  document.getElementById('btn-guardar-sugerencias').addEventListener('click', ()=>{
   967→    lastSuggestions.forEach((s,i)=>{
   968→      const checkEl = document.getElementById(`sugg-check-${i}`);
   969→      if(!checkEl || !checkEl.checked) return;
   970→      const cat = up(document.getElementById(`sugg-cat-${i}`).value.trim());
   971→      const syns = document.getElementById(`sugg-syn-${i}`).value.split(',').map(x=>up(x.trim())).filter(Boolean);
   972→      if(!cat) return;
   973→      const existing = customDict[cat] || PARTS_TO_CHECK[cat] || [];
   974→      customDict[cat] = [...new Set([...existing, ...syns, cat])];
   975→    });
   976→    saveCustomDict(customDict);
   977→    document.getElementById('suggestions-box').innerHTML = '<div class="warn">Diccionario actualizado. Corriendo de nuevo la auditoría con las categorías nuevas...</div>';
   978→    lastSuggestions = [];
   979→    runAudit();
   980→  });
   981→}
