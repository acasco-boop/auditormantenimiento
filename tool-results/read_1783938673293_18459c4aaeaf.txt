   370→        headers.forEach((h, i)=>{

   375→        });
   376→        rows.push(obj);

   378→    });
   379→    return rows;
   380→  }

   492→  if(rows.length === 0) return {ok:false};
   493→  const cols = Object.keys(rows[0]);
   494→  if(cols.includes(primaryCol)){

   503→function runAudit(){
   504→  const errBox = document.getElementById('load-status');
   505→  errBox.innerHTML = '';
   506→
   507→  if(!dfTar || !dfMat){ return; }
   508→
   509→  const okTar = normalizeOrderCol(dfTar, 'DocNum');
   510→  if(!okTar.ok){
   511→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (DocNum o Nro. Orden) en el archivo de Tareas.</div>';
   512→    return;
   513→  }
   514→  const okMat = normalizeOrderCol(dfMat, 'Nro. OM');
   515→  if(!okMat.ok){
   516→    errBox.innerHTML = '<div class="err">No se encontró la columna de Orden (Nro. OM o Nro. Orden) en el archivo de Materiales.</div>';
   517→    return;
   518→  }
   519→
   520→  // Materiales indexados por Nro. Orden (global para desglose en métricas)
   521→  matByOrder = {};
   522→  dfMat.forEach(r=>{
   523→    const key = String(r['Nro. Orden']);
   524→    if(!matByOrder[key]) matByOrder[key] = [];
   525→    matByOrder[key].push(r);
   526→  });
   527→
   528→  const activeParts = getActiveParts();
   529→  const unrecognizedMap = {};
   530→
   531→  results = [];
   532→  dfTar.forEach(row=>{
   533→    const tarea = row['Tarea'] === null || row['Tarea'] === undefined ? '' : String(row['Tarea']);
   534→    const tareaUp = up(tarea);
   535→    if(!explicitReplacementNeeded(tareaUp)) return;
   536→
   537→    const order = row['Nro. Orden'];
   538→
   539→    // Detección de repuestos aún no reconocidos por el diccionario (de fábrica
   540→    // + aprendido). Se registra independientemente del cruce con materiales,
   541→    // porque el objetivo es identificar vocabulario nuevo para poder sumarlo.
   542→    const matchesKnownCategory = Object.keys(activeParts).some(kw => tareaUp.includes(kw));
   543→    if(!matchesKnownCategory){
   544→      const key = tareaUp;
   545→      if(!unrecognizedMap[key]) unrecognizedMap[key] = {tarea, count:0, order};
   546→      unrecognizedMap[key].count++;
   547→    }
   548→
   549→    const mats = matByOrder[String(order)] || [];
   550→    const matList = mats.map(m => up(m['Desc. Artículo']));
   551→    const totalSalidas = mats.reduce((acc,m)=>{
   552→      const v = parseFloat(m['Salidas']);
   553→      return acc + (isNaN(v) ? 0 : v);
   554→    }, 0);
   555→
   556→    const base = {
   557→      'Nro. Orden': order,
   558→      'Equipo': row['Codigo equipo'],
   559→      'Nombre Equipo': row['Nombre Equipo'],
   560→      'Tarea': tarea,
   561→      'Estado Tarea': row['Estado']
   562→    };
   563→
   564→    if(mats.length === 0){
   565→      results.push({...base,
   566→        'Tipo de Hallazgo': '1) Orden sin repuestos asignados',
   567→        'Detalle': 'Ningún material cargado en la orden'
   568→      });
   569→    } else if(totalSalidas === 0){
   570→      results.push({...base,
   571→        'Tipo de Hallazgo': '3) Planificado sin Salida física (Salidas = 0)',
   572→        'Detalle': 'Materiales planificados con Salida Cero: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   573→      });
   574→    } else {
   575→      // Se revisan TODAS las categorías mencionadas en la tarea (no se corta
   576→      // en la primera que matchea), así una tarea que pide varios repuestos
   577→      // (ej. "Cambiar filtro y correa") no queda "aprobada" con que solo uno
   578→      // de los dos haya sido cargado en el pañol.
   579→      const missingCategories = [];
   580→      for(const kw in activeParts){
   581→        if(tareaUp.includes(kw)){
   582→          const matsKws = activeParts[kw];
   583→          const found = matList.some(m => matsKws.some(mk => m.includes(mk)));
   584→          if(!found) missingCategories.push(kw);
   585→        }
   586→      }
   587→      if(missingCategories.length > 0){
   588→        results.push({...base,
   589→          'Tipo de Hallazgo': `2) Desconexión de material (Falta '${missingCategories.join("', '")}')`,
   590→          'Detalle': 'Materiales cargados no coinciden: ' + mats.map(m=>m['Desc. Artículo']).join(', ')
   591→        });
   592→      }
   593→    }
   594→  });
   595→
   596→  unrecognizedTasks = Object.values(unrecognizedMap).sort((a,b)=>b.count-a.count);
   597→
   598→  renderResults();
   599→  renderUnrecognized();
   600→  renderCustomDictList();
   601→}

   603→/* ============ Render de detalle por métrica (OM únicas + almacén) ============ */
   604→function renderMetricDetail(elId, prefix, count){
   605→  const el = document.getElementById(elId);
   606→  if(!el) return;
   607→
   608→  const subset = results.filter(r => r['Tipo de Hallazgo'].startsWith(prefix));
   609→
   610→  if(count === 0){
   611→    el.innerHTML = '';
   612→    return;
   613→  }
   614→
   615→  // OM únicas
   616→  const uniqueOMs = [...new Set(subset.map(r => r['Nro. Orden']))];
   617→  const omCount = uniqueOMs.length;
   618→
   619→  let html = '';
   620→  html += '<div class="detail-row"><span class="label">Tareas afectadas</span><span class="value">' + count + '</span></div>';
   621→  html += '<div class="detail-row"><span class="label">OM únicas</span><span class="value">' + omCount + '</span></div>';
   622→
   623→  // Para tipo 1: sin materiales, no hay almacén
   624→  if(prefix === '1)'){
   625→    html += '<div class="warehouse-title">Desglose por almacén</div>';
   626→    html += '<div class="na-msg">No aplica (0 repuestos)</div>';
   627→    el.innerHTML = html;
   628→    return;
   629→  }
   630→
   631→  // Para tipos 2 y 3: contar materiales por almacén de las OM afectadas
   632→  const whCount = {};
   633→  uniqueOMs.forEach(om => {
   634→    const mats = matByOrder[String(om)] || [];
   635→    mats.forEach(m => {
   636→      const wh = String(m['Almacen'] || 'S/D').trim();
   637→      if(!wh) return;
   638→      whCount[wh] = (whCount[wh] || 0) + 1;
   639→    });
   640→  });
   641→
   642→  const whEntries = Object.entries(whCount).sort((a,b) => b[1] - a[1]);
   643→  const maxWh = whEntries.length > 0 ? whEntries[0][1] : 1;
   644→
   645→  html += '<div class="warehouse-title">Desglose por almacén</div>';
   646→  whEntries.forEach(([wh, cnt]) => {
   647→    const pct = Math.round((cnt / maxWh) * 100);
   648→    html += '<div class="wh-row">'
   649→      + '<span class="wh-name">' + esc(wh) + '</span>'
   650→      + '<div class="wh-bar-wrap"><div class="wh-bar" style="width:' + pct + '%"></div></div>'
   651→      + '<span class="wh-count">' + cnt + '</span>'
   652→      + '</div>';
   653→  });
   654→
   655→  if(whEntries.length === 0){
   656→    html += '<div class="na-msg">Sin datos de almacén</div>';
   657→  }
   658→
   659→  el.innerHTML = html;
   660→}
