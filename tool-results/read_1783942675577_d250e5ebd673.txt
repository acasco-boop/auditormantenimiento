    41→    41→async function parseTabularFile(file: File): Promise<Record<string, unknown>[]> {
    42→    42→  const name = file.name.toLowerCase();
    43→    43→  if (name.endsWith('.csv')) {
    44→    44→    const text = await file.text();
    45→    45→    const parsed = Papa.parse(text, { header: true, skipEmptyLines: true, dynamicTyping: true });
    46→    46→    return parsed.data as Record<string, unknown>[];
    47→    47→  } else {
    48→    48→    const buf = await file.arrayBuffer();
    49→    49→    const wb = new ExcelJS.Workbook();
    50→    50→    await wb.xlsx.load(buf);
    51→    51→    const ws = wb.worksheets[0];
    52→    52→    const rows: Record<string, unknown>[] = [];
    53→    53→    let headers: string[] = [];
    54→    54→    ws.eachRow((row, rowNumber) => {
    55→    55→      const values = row.values.slice(1);
    56→    56→      if (rowNumber === 1) {
    57→    57→        headers = values.map(v => (v === undefined || v === null) ? '' : String(v).trim());
    58→    58→      } else {
    59→    59→        const obj: Record<string, unknown> = {};
    60→    60→        headers.forEach((h, i) => {
    61→    61→          let v = values[i];
    62→    62→          if (v && typeof v === 'object' && 'text' in (v as object)) v = (v as { text: unknown }).text;
    63→    63→          if (v && typeof v === 'object' && 'result' in (v as object)) v = (v as { result: unknown }).result;
    64→    64→          obj[h] = v === undefined ? null : v;
    65→    65→        });
    66→    66→        rows.push(obj);
    67→    67→      }
    68→    68→    });
    69→    69→    return rows;
    70→    70→  }
    71→    71→}

   692→   692→function SuggestionCard({ suggestion, index, onUpdate }: { suggestion: AISuggestion; index: number; onUpdate: (updated: AISuggestion) => void }) {
   693→   693→  const [cat, setCat] = useState(up(suggestion.categoria || ''));
   694→   694→  const [syns, setSyns] = useState((suggestion.sinonimos || []).map(x => up(x)).join(', '));
   695→   695→  const [checked, setChecked] = useState(true);
   696→   696→
   697→   697→  useEffect(() => {
   698→   698→    onUpdate({ ...suggestion, categoria: cat, sinonimos: syns.split(',').map(x => x.trim()).filter(Boolean) });
   699→   699→  }, [cat, syns]);
   700→   700→
   701→   701→  return (
   702→   702→    <div className="glass-card rounded-xl overflow-hidden">
   703→   703→      <div className="p-4">
   704→   704→        <p className="font-mono text-xs text-slate-400 mb-3 break-words">&quot;{esc(suggestion.tarea)}&quot;</p>
   705→   705→        <div className="grid grid-cols-[auto_1fr] gap-3 items-center">
   706→   706→          <Checkbox checked={checked} onCheckedChange={(v) => setChecked(!!v)} className="data-[state=checked]:bg-amber-500 data-[state=checked]:border-amber-500" />
   707→   707→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   708→   708→            <Input value={cat} onChange={e => setCat(up(e.target.value))} placeholder="CATEGORÍA" className="font-mono text-xs bg-white/[0.03] border-white/[0.06] h-8 rounded-lg" />
   709→   709→            <Input value={syns} onChange={e => setSyns(e.target.value)} placeholder="Sinónimos separados por coma" className="font-mono text-xs bg-white/[0.03] border-white/[0.06] h-8 rounded-lg" />
   710→   710→          </div>
   711→   711→        </div>
   712→   712→      </div>
   713→   713→    </div>
   714→   714→  );
